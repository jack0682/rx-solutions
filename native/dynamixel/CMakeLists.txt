cmake_minimum_required(VERSION 3.20)
project(rx_dynamixel_ping LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "DXL_LINUX_HOST_CHANNEL_REQUIRED")
endif()
find_package(nlohmann_json 3.10 REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/source.json" pins)
string(JSON count LENGTH "${pins}" files)
math(EXPR last "${count} - 1")
foreach(i RANGE ${last})
  string(JSON name MEMBER "${pins}" files ${i})
  string(JSON expected GET "${pins}" files "${name}" sha256)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${name}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "DYNAMIXEL upstream content pin differs: ${name}")
  endif()
endforeach()
add_executable(rx-dynamixel-ping helper.cpp vendor/protocol2_packet_handler.cpp)
target_include_directories(rx-dynamixel-ping PRIVATE vendor)
target_link_libraries(rx-dynamixel-ping PRIVATE nlohmann_json::nlohmann_json)
target_compile_features(rx-dynamixel-ping PRIVATE cxx_std_20)
target_compile_options(rx-dynamixel-ping PRIVATE -Wall -Wextra -Werror)
install(TARGETS rx-dynamixel-ping RUNTIME DESTINATION bin)
install(FILES vendor/LICENSE DESTINATION licenses/DynamixelSDK)
