cmake_minimum_required(VERSION 3.22)
project(rx_bt_executor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BTCPP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/BehaviorTree.CPP" CACHE PATH "Pinned BT.CPP source")
set(BTCPP_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BTCPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BTCPP_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BTCPP_GROOT_INTERFACE OFF CACHE BOOL "" FORCE)
set(BTCPP_SQLITE_LOGGING OFF CACHE BOOL "" FORCE)
add_subdirectory("${BTCPP_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/btcpp")
add_library(rx_bt_executor runtime.cpp nodes.cpp xml.cpp frame.cpp)
target_include_directories(rx_bt_executor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${BTCPP_SOURCE}/3rdparty/tinyxml2")
target_link_libraries(rx_bt_executor PUBLIC behaviortree_cpp PRIVATE tinyxml2)
target_compile_options(rx_bt_executor PRIVATE -Wall -Wextra -Werror)
add_executable(rx-bt-tests tests.cpp)
target_link_libraries(rx-bt-tests PRIVATE rx_bt_executor)
target_compile_options(rx-bt-tests PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME rx-bt-contract COMMAND rx-bt-tests)
add_executable(rx-bt-engine engine.cpp)
target_link_libraries(rx-bt-engine PRIVATE rx_bt_executor)
target_compile_options(rx-bt-engine PRIVATE -Wall -Wextra -Werror)
option(RX_BUILD_TEST_HARNESS "Build simulation-only process fixtures" OFF)
if(RX_BUILD_TEST_HARNESS)
add_executable(rx-bt-engine-fixture engine.cpp)
target_compile_definitions(rx-bt-engine-fixture PRIVATE RX_BT_TEST_CLOCK=1)
target_link_libraries(rx-bt-engine-fixture PRIVATE rx_bt_executor)
target_compile_options(rx-bt-engine-fixture PRIVATE -Wall -Wextra -Werror)
# This executable deliberately supports synthetic clocks and is only a validation target.
add_executable(rx-bt-request-fixture request_fixture.cpp)
target_link_libraries(rx-bt-request-fixture PRIVATE rx_bt_executor)
target_compile_options(rx-bt-request-fixture PRIVATE -Wall -Wextra -Werror)

endif()
