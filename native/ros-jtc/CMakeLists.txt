cmake_minimum_required(VERSION 3.22)
project(rx_ros_jtc_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(control_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(action_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
set(RX_CATALOG "${CMAKE_CURRENT_SOURCE_DIR}/../../catalogs/device-support.v1.json" CACHE FILEPATH "Pinned device support catalog")
file(READ "${RX_CATALOG}" RX_CATALOG_JSON)
file(SHA256 "${RX_CATALOG}" RX_CATALOG_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RX_CATALOG}")
configure_file(catalog.hpp.in catalog.hpp @ONLY)
add_executable(rx-ros-jtc-bridge bridge.cpp)
target_include_directories(rx-ros-jtc-bridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
ament_target_dependencies(rx-ros-jtc-bridge rclcpp control_msgs controller_manager_msgs action_msgs)
target_link_libraries(rx-ros-jtc-bridge nlohmann_json::nlohmann_json)
target_compile_options(rx-ros-jtc-bridge PRIVATE -Wall -Wextra -Werror)
install(TARGETS rx-ros-jtc-bridge DESTINATION bin)
